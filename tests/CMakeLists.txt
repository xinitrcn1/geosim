function(geosim_add_test name lib)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE ${lib})
    target_compile_definitions(${name} PRIVATE GEOSIM_ASSETS_DIR="${PROJECT_SOURCE_DIR}/assets")
    geosim_target_defaults(${name})
    if(lib STREQUAL "geosim_sim")
        target_precompile_headers(${name} REUSE_FROM geosim_sim) # the generated container, parsed once
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "unit")
endfunction()
